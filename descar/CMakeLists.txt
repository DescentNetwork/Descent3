cmake_minimum_required(VERSION 3.0)

# set default cmake build type to Debug (None Debug Release RelWithDebInfo MinSizeRel)
if(NOT CMAKE_BUILD_TYPE AND NOT DEFINED ENV{CMAKE_BUILD_TYPE})
  set(CMAKE_BUILD_TYPE "Debug" CACHE STRING "default build type")
endif()

project(descar
  LANGUAGES C CXX
  VERSION 0.1.0)

option(FORCE_COLORED_OUTPUT "Always produce ANSI-colored compiler warnings/errors (GCC/Clang only; esp. useful with ninja)." OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(FORCE_COLORED_OUTPUT)
  if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
    set(CMAKE_COLOR_DIAGNOSTICS ON)
  else()
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      add_compile_options (-fdiagnostics-color=always)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      add_compile_options (-fcolor-diagnostics)
    endif()
  endif()
endif()

set(REPO_GIT_HASH "")
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(
    COMMAND ${GIT_EXECUTABLE} describe --always
    OUTPUT_VARIABLE REPO_GIT_HASH
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
  string(PREPEND "git hash " REPO_GIT_HASH)
endif()

string(TOUPPER ${PROJECT_NAME} MACRO_PREFIX)
file(WRITE ${PROJECT_BINARY_DIR}/version.h
"#pragma once
/* autogenerated file */

#define ${MACRO_PREFIX}_MAJORVER   ${PROJECT_VERSION_MAJOR}
#define ${MACRO_PREFIX}_MINORVER   ${PROJECT_VERSION_MINOR}
#define ${MACRO_PREFIX}_BUILD      ${PROJECT_VERSION_PATCH}
#define ${MACRO_PREFIX}_GIT_HASH  \"${REPO_GIT_HASH}\"
")

set(UTILITIES
  utils/fixed_string.h
  utils/string_hash.h
  utils/portable_endian.h
  utils/posix_stream.h
  utils/posix_stream.cpp)

set(HEADERS
  src/hog2_format.h
  ${UTILITIES})

set(CPPS
  src/hog2_format.cpp
  src/main.cpp)

add_executable(descar ${HEADERS} ${CPPS})
target_include_directories(descar PRIVATE ${PROJECT_BINARY_DIR} utils)
